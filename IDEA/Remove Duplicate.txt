#include <vector>
using namespace std;

void RemoveDuplicates(Node* root, vector<int>& v)
{
    if (root == nullptr)
        return;

    RemoveDuplicates(root->left, v);

    bool found = false;

    for (int i = 0; i < v.size(); i++)
    {
        if (v[i] == root->data)
        {
            found = true;
            break;
        }
    }

    if (found)
        Delete(root->data);   // دالة الحذف في BST
    else
        v.push_back(root->data);

    RemoveDuplicates(root->right, v);
}
