#include <iostream>
using namespace std;

class Node {
public:
    int data;
    Node* left;
    Node* right;

    Node(int val) {
        data = val;
        left = right = nullptr;
    }
};

class BST {
private:
    Node* root;

    Node* Insert(Node* node, int val) {
        if (node == nullptr)
            return new Node(val);

        if (val < node->data)
            node->left = Insert(node->left, val);
        else
            node->right = Insert(node->right, val);

        return node;
    }

    void InOrder(Node* node) {
        if (node == nullptr)
            return;

        InOrder(node->left);
        cout << node->data << " ";
        InOrder(node->right);
    }

public:
    BST() {
        root = nullptr;
    }

    void Insert(int val) {
        root = Insert(root, val);
    }

    Node* GetRoot() {
        return root;
    }

    void Print() {
        InOrder(root);
        cout << endl;
    }
};

void StoreEvenOdd(Node* root, BST& evenTree, BST& oddTree) {
    if (root == nullptr)
        return;

    if (root->data % 2 == 0)
        evenTree.Insert(root->data);
    else
        oddTree.Insert(root->data);

    StoreEvenOdd(root->left, evenTree, oddTree);
    StoreEvenOdd(root->right, evenTree, oddTree);
}

int main() {

    BST original;
    BST evenTree;
    BST oddTree;

    original.Insert(10);
    original.Insert(5);
    original.Insert(15);
    original.Insert(8);
    original.Insert(3);
    original.Insert(20);
    original.Insert(7);

    StoreEvenOdd(original.GetRoot(), evenTree, oddTree);

    cout << "Even Tree: ";
    evenTree.Print();

    cout << "Odd Tree: ";
    oddTree.Print();

    return 0;
}
